```ts
export type SpeechAvailability = 'ok' | 'insecure' | 'unsupported' | 'denied'
export type SpeechVerdict = 'pass' | 'fail' | 'no_speech' | 'error'

export type SpeechResult = {
  verdict: SpeechVerdict
  transcript: string
  confidence: number
  candidates: string[]
}

export type SpeechInput = {
  transcript: string
  confidence: number
  candidates?: readonly string[]
}

export const AVAILABILITY_REASON: Record<Exclude<SpeechAvailability, 'ok'>, string> = {
  insecure: '请用 https 打开',
  unsupported: '此浏览器不支持语音，请用 Chrome/Edge',
  denied: '麦克风权限被拒，请点地址栏左侧图标开启'
}

export function resolveAvailability(input: {
  hasRecognition: boolean
  secure: boolean
  denied: boolean
}): SpeechAvailability {
  if (!input.hasRecognition) return 'unsupported'
  if (!input.secure) return 'insecure'
  if (input.denied) return 'denied'
  return 'ok'
}

/** §8.3 第 1 步：trim + 小写 + 去标点 + 折叠空格 */
export function normalizeSpeech(text: string): string {
  return text
    .normalize('NFKC')
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, '')
    .replace(/\s+/g, ' ')
    .trim()
}

export function editDistance(a: string, b: string): number {
  if (a === b) return 0
  if (!a.length) return b.length
  if (!b.length) return a.length
  let prev: number[] = Array.from({ length: b.length + 1 }, (_, i) => i)
  for (let i = 1; i <= a.length; i++) {
    const cur: number[] = [i]
    for (let j = 1; j <= b.length; j++) {
      cur[j] = Math.min(
        prev[j] + 1,
        cur[j - 1] + 1,
        prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1)
      )
    }
    prev = cur
  }
  return prev[b.length]
}

/** §8.3：顺序固定；pass 判定先于 no_speech 阈值（ASR 置信度为 0 时仍可能读对） */
export function judgeTranscript(expect: string, input: SpeechInput): SpeechVerdict {
  const target = normalizeSpeech(expect)
  const pool = (input.candidates?.length ? input.candidates : [input.transcript])
    .slice(0, 5)
    .map(normalizeSpeech)

  if (pool.some((c) => c.length > 0 && c === target)) return 'pass'
  if (target.length >= 5 && pool.some((c) => c.length > 0 && editDistance(c, target) <= 1)) return 'pass'
  if (input.transcript.trim() === '' || input.confidence < 0.4) return 'no_speech'
  return 'fail'
}
```

```ts
import {
  judgeTranscript,
  resolveAvailability,
  type SpeechAvailability,
  type SpeechResult
} from '../rules/speechJudge'

export type RecognizeOptions = { timeoutMs: number }

export interface SpeechProvider {
  available(): SpeechAvailability
  recognize(expect: string, opts: RecognizeOptions): Promise<SpeechResult>
  refreshPermission?(): Promise<SpeechAvailability>
}

type Alternative = { transcript?: string; confidence?: number }
type RecognitionResult = { length: number; [index: number]: Alternative }
type RecognitionEvent = { results?: { length: number; [index: number]: RecognitionResult } }
type RecognitionErrorEvent = { error?: string }

type RecognitionLike = {
  lang: string
  continuous: boolean
  interimResults: boolean
  maxAlternatives: number
  start(): void
  stop(): void
  abort(): void
  onresult: ((event: RecognitionEvent) => void) | null
  onerror: ((event: RecognitionErrorEvent) => void) | null
  onend: (() => void) | null
}

type RecognitionCtor = new () => RecognitionLike

const emptyResult = (verdict: SpeechResult['verdict']): SpeechResult => ({
  verdict,
  transcript: '',
  confidence: 0,
  candidates: []
})

function getCtor(): RecognitionCtor | undefined {
  if (typeof window === 'undefined') return undefined
  const w = window as unknown as {
    SpeechRecognition?: RecognitionCtor
    webkitSpeechRecognition?: RecognitionCtor
  }
  return w.SpeechRecognition ?? w.webkitSpeechRecognition
}

export class NoopProvider implements SpeechProvider {
  available(): SpeechAvailability {
    return 'unsupported'
  }
  async recognize(_expect: string, _opts: RecognizeOptions): Promise<SpeechResult> {
    return emptyResult('error')
  }
}

export class WebSpeechProvider implements SpeechProvider {
  private denied = false

  available(): SpeechAvailability {
    const secure = typeof window === 'undefined' ? false : window.isSecureContext
    return resolveAvailability({
      hasRecognition: getCtor() !== undefined,
      secure,
      denied: this.denied
    })
  }

  async refreshPermission(): Promise<SpeechAvailability> {
    try {
      if (typeof navigator !== 'undefined' && navigator.permissions) {
        const status = await navigator.permissions.query({
          name: 'microphone' as PermissionName
        })
        this.denied = status.state === 'denied'
      }
    } catch {
      // 部分浏览器不支持 microphone 查询：保持现有状态
    }
    return this.available()
  }

  recognize(expect: string, opts: RecognizeOptions): Promise<SpeechResult> {
    return new Promise((resolve) => {
      const Ctor = getCtor()
      if (!Ctor) {
        resolve(emptyResult('error'))
        return
      }

      const rec = new Ctor()
      rec.lang = 'en-US'
      rec.continuous = false
      rec.interimResults = false
      rec.maxAlternatives = 5

      let done = false
      const finish = (result: SpeechResult) => {
        if (done) return
        done = true
        globalThis.clearTimeout(timer)
        rec.onresult = null
        rec.onerror = null
        rec.onend = null
        try {
          rec.abort()
        } catch {
          // 已经停止
        }
        resolve(result)
      }

      const timer = globalThis.setTimeout(() => finish(emptyResult('no_speech')), opts.timeoutMs)

      rec.onresult = (event) => {
        const first = event.results?.[0]
        if (!first || first.length === 0) {
          finish(emptyResult('no_speech'))
          return
        }
        const alternatives: Alternative[] = []
        for (let i = 0; i < first.length && i < 5; i++) {
          alternatives.push({
            transcript: String(first[i]?.transcript ?? ''),
            confidence: Number(first[i]?.confidence ?? 0)
          })
        }
        const transcript = alternatives[0].transcript ?? ''
        const confidence = alternatives[0].confidence ?? 0
        const candidates = alternatives.map((a) => a.transcript ?? '')
        finish({
          verdict: judgeTranscript(expect, { transcript, confidence, candidates }),
          transcript,
          confidence,
          candidates
        })
      }

      rec.onerror = (event) => {
        const code = event.error ?? ''
        if (code === 'not-allowed' || code === 'service-not-allowed') this.denied = true
        finish(emptyResult(code === 'no-speech' || code === 'aborted' ? 'no_speech' : 'error'))
      }

      rec.onend = () => finish(emptyResult('no_speech'))

      try {
        rec.start()
      } catch {
        finish(emptyResult('error'))
      }
    })
  }
}

export function createSpeechProvider(): SpeechProvider {
  return getCtor() ? new WebSpeechProvider() : new NoopProvider()
}
```

```ts
import { describe, expect, it } from 'vitest'
import { CTX as ctx, M0_LEVEL_ID } from '../../src/content/index'
import { focusCard } from '../../src/rules/cards'
import { SPEECH_BUFF } from '../../src/rules/constants'
import { onCellClick, onTileClick } from '../../src/rules/play'
import { newGame } from '../../src/rules/sim'
import { expireVoiceBuffs, grantVoicePass, isVoiceBuffed, voiceBuffFields } from '../../src/rules/voice'
import type { GameState } from '../../src/rules/types'

function armFocus(state: GameState): GameState {
  let s = state
  for (;;) {
    const card = focusCard(s.cards)
    if (!card || card.status !== 'filling') return s
    const need = card.slots[card.filled]
    const tile = card.tiles.find((t) => !t.used && t.letter === need)!
    s = onTileClick(s, tile.tileId, ctx).state
  }
}

const withSun = (s: GameState): GameState => ({ ...s, sun: 9990 })

describe('卡片通过标记', () => {
  it('grantVoicePass 只标记目标卡，且不可重复标记', () => {
    const s0 = newGame(M0_LEVEL_ID, ctx)
    const id = focusCard(s0.cards)!.instanceId
    const s1 = grantVoicePass(s0, id)
    expect(focusCard(s1.cards)!.voiceBuffed).toBe(true)
    expect(grantVoicePass(s1, id)).toBe(s1)
    expect(grantVoicePass(s0, 9999)).toBe(s0)
  })
})

describe('§8.4 种植时挂 buff', () => {
  it('通过过的卡种下后携带 10s buff 与 30s 冷却，并发 voice_buff', () => {
    const s0 = withSun(armFocus(newGame(M0_LEVEL_ID, ctx)))
    const id = focusCard(s0.cards)!.instanceId
    const s1 = grantVoicePass(s0, id)
    const r = onCellClick(s1, 2, 3, ctx)
    const plant = r.state.plants[0]
    expect(plant.buffUntil).toBe(0 + SPEECH_BUFF.duration_s)
    expect(plant.voiceReadyAt).toBe(0 + SPEECH_BUFF.cooldown_s)
    expect(isVoiceBuffed(plant, 0)).toBe(true)
    expect(isVoiceBuffed(plant, SPEECH_BUFF.duration_s)).toBe(false)
```

```ts
/** 读音通过：只标记，不直接改植物（植物还没种下） */
export function grantVoicePass(state: GameState, instanceId: number): GameState {
  const card = state.cards.find((c) => c.instanceId === instanceId)
  if (!card || card.voiceBuffed || card.status === 'placed' || card.status === 'dormant') return state
  return {
    ...state,
    cards: state.cards.map((c) => (c.instanceId === instanceId ? { ...c, voiceBuffed: true } : c))
  }
}

/** §8.4 全局至多 1 个：新加成开始时让旧的立即到期 */
export function expireVoiceBuffs(plants: readonly Plant[], now: number): Plant[] {
  return plants.map((p) =>
    p.buffUntil !== undefined && p.buffUntil > now ? { ...p, buffUntil: now } : p
  )
}

export function isVoiceBuffed(plant: Plant, now: number): boolean {
  return plant.buffUntil !== undefined && plant.buffUntil > now

  return {
    buffUntil: now + SPEECH_BUFF.duration_s,
    voiceReadyAt: now + SPEECH_BUFF.cooldown_s

}
```

1. **超时判定为 `no_speech`**：§8.2 只规定「8 秒硬超时自动停」，未规定 verdict；按「没听清」处理可免费重试，不记错词。
2. **候选命中先于置信度阈值**：严格按 §8.3 的步骤顺序实现，因此 `confidence < 0.40` 但候选完全命中时仍判 pass（ASR 常把正确读法排在候选里、置信度报 0）；已用单测锁定。
3. **armed 卡片仍可跟读**：孩子可以先拼完再读，通过后同样在种下时生效；`placed/dormant` 卡片拒绝标记。
4. **30 秒冷却为数据级**：`voiceReadyAt = 种植时刻 + 30` 写入植物；当前 UI 每张卡通过后按钮即锁定、种下后卡片离开手牌，不存在同一株二次加成的入口，因此未额外实现运行时拒绝分支。
5. **语音 fail 只进存档 `wrong`**：结算页「点错字母」仍只统计字母块填错，避免把发音错误混进拼写统计；与 §8.4「只有 fail 记一次尝试」一致。
6. **`fireShooters` 的 `now` 参数可选**（默认 0），兼容既有 4 参调用与测试；`sim.ts` 显式传 `t`。
7. **四态优先级**：`unsupported > insecure > denied`。浏览器没有识别能力时先报浏览器不支持，避免建议用户改用 https 却仍然不可用。
8. **设置入口放在选关页**：本里程碑只做 TTS 开关与隐私声明；`penalty_disable_ms` 可调仍留待后续，不扩大 M3 范围。
9. **未新增存档字段**：语音失败复用 `word_stats.wrong`，不升级 `SAVE_VERSION`。
10. **真实麦克风未验**：本轮用假 ASR 覆盖链路与四态；E11 五种识别环境必须人工执行。
